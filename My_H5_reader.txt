        function updateSeriesDropdown(app)
            % Находим все оставшиеся линии
            lines = findobj(app.UIAxes, 'Type', 'line');
            
            if ~isempty(lines)
                % Обновляем DropDown
                lineNames = arrayfun(@(x) x.DisplayName, lines, 'UniformOutput', false);
                app.DropDown.Items = lineNames;
                %app.DropDown.ItemsData = lines; % (если используется ItemsData)
            else
                % Если линий не осталось — очищаем DropDown
                app.DropDown.Items = {'NA'};
                app.DropDown.Enable=0;
                app.SeriesclearButton.Enable=0;
                %app.CheckStimulBox.Enable=0;
                app.CheckStimulBox.Value=0;
                app.ClearChartButton.Enable=0;
                %app.DropDown.ItemsData = [];
            end
end
        
        function SortTable(app, Table, RowXIndex)
            cell_style = uistyle('BackgroundColor',[1 0.6 0.6]);
            data = Table.Data; % Извлекаем данные из таблицы
            columnNames=Table.ColumnName;
            rowNames = data(:, 1); % Извлекаем названия рядов (первый столбец)
            
            % Извлекаем данные для сортировки (все столбцы, кроме первого)
            dataToSort = data(:, 2:end); % Игнорируем первый столбец
            rowToSort=dataToSort(RowXIndex, :);
            
            % Сортируем данные по первой строке
            % Мы используем sortrows для сортировки в зависимости от первой строки данных
            Xc=table2cell(rowToSort);
            Xa=cell2mat(Xc);
            [~, sortOrder]=sort(Xa); % Получаем порядок сортировки по первой строке
    
            % Переставляем столбцы в соответствии с порядком сортировки
            sortedData = data(:, [1, sortOrder + 1]); % Сохраняем первый столбец и переставляем остальные
            sortedCoulmnNames=[columnNames{1}, columnNames(sortOrder+1)']; 
          
            % Обновляем данные таблицы
            Table.Data = sortedData;
             % Обновляем заголовки столбцов
            Table.ColumnName=sortedCoulmnNames;
            addStyle(Table,cell_style,'row',[RowXIndex]);
        end
   
        function showMessage(app, Label)
            Label.Visible=1;     
            t = timer; % Создайте таймер
            t.StartDelay = 1; % Таймер с ожиданием 1 секунды
            t.TimerFcn = @(~, ~) closeAlert(app, t, Label); % Установите функцию закрытия
            start(t); % Запустите таймер
         end
        
        % Функция для закрытия предупреждения
        function closeAlert(app, t, Label)
            stop(t); % Остановить таймер
            delete(t); % Удалить таймер
            Label.Visible=0;    
        end
        
        function SavetabletoXLS(app, Table, capture)
           data = Table.Data;          
           dateTime = datestr(now, 'yyyymmdd_HHMMSS'); 
           save_pname=app.FolderMenu.Tooltip;
           fileName = sprintf(capture, dateTime); 
           Outputfile=strcat(save_pname, fileName);
           writetable(data, Outputfile);
           winopen(Outputfile);   
        end
        
        function DelCol (app, Table, DropDown)
           columnToDel=DropDown.Value;  
           Table.Data(:, strcmp(Table.ColumnName, columnToDel))=[];
           Table.ColumnName(strcmp(Table.ColumnName, columnToDel))=[];
           itemIndex = find(strcmp(DropDown.Items, columnToDel));
           DropDown.Items(itemIndex) = [];
           if ~isempty(DropDown.Items)
              DropDown.Value = DropDown.Items{1}; 
           else
              Table.Data(:,1)=[];
              Table.ColumnName(1)=[];
           end      
        end       
        
               
        function sort_flag_new = DrawChart (app, Axes, DropDown, XDropDown, Table, Label, sort_flag)         
            selected_Value=DropDown.Value;    
            if contains(selected_Value,"_")
                if contains(selected_Value," ")
                     title_selected_Value=strcat(extractBefore(selected_Value,"_"),"_{",extractBetween(selected_Value,"_"," "),"}");
                else
                     title_selected_Value=strcat(extractBefore(selected_Value,"_"),"_{",extractAfter(selected_Value,"_"),"}");
                end    
            else
                title_selected_Value=selected_Value;      
            end
            if contains(selected_Value,"(")
                Ytitle=strcat(extractBetween(selected_Value,"(",")"));
            else 
                Ytitle=" ";
            end
            Axes.Title.String=title_selected_Value;
            Axes.YLabel.String=Ytitle;
            selected_XAxis=XDropDown.Value;  
            Axes.XLabel.String=selected_XAxis;
            
            data = Table.Data; 
            rowNames = data(:, 1);
            Yc=table2cell(rowNames);
        
            rowXIndex=find(strcmp(Yc, selected_XAxis));
            if rowXIndex ~= sort_flag
                 if sort_flag~=0   
                     cell_style = uistyle('BackgroundColor',[1 1 1]);
                     addStyle(Table,cell_style,'row',[sort_flag]);   
                 end
                     SortTable(app, Table, rowXIndex); 
                     showMessage(app, Label);
                     sort_flag_new=rowXIndex;
            else  
                 sort_flag_new=sort_flag;
            end
        
            data = Table.Data; 
            DataToDraw = data(:, 2:end); %already resorted table
            
            rowIndex=find(strcmp(Yc, selected_Value));
            
            XrowToDraw=DataToDraw(rowXIndex, :);
            Xc=table2cell(XrowToDraw);
            Xa=cell2mat(Xc);
        
            if numel(unique(Xa)) < numel(Xa)
                 uialert(app.MainUIFigure,['The same frequencies in the columns of the table.' ...
                   ' It is impossible to draw a graph.  Please delete one of the columns.'],...
                   'Same Frequences');   
            else
                 if rowXIndex==1 
                     Axes.XLim=[0.3, max(Xa)+0.3];
                 else
                     Axes.XLim=[100, max(Xa)+100];
                 end    
                Axes.XTick=Xa;
          
                YrowToDraw=DataToDraw(rowIndex, :);
                Yc=table2cell(YrowToDraw);
                Ya=cell2mat(Yc);
                Axes.YLim=[min(Ya)-0.1*(max(Ya)-min(Ya)), max(Ya)+0.1*(max(Ya)-min(Ya))];
                
                plot(Axes,Xa,Ya,'*-','LineWidth',2,'Markersize',7,'MarkerEdgeColor','red');  
            end 
        end 
    end

        % Value changed function: UpSlider
        function UpSliderValueChanged(app, event)
         global cycle
            value = app.UpSlider.Value;
            if value>=app.DownSlider.Value 
                app.UIAxes.XLim=[app.DownSlider.Value-cycle, app.DownSlider.Value];
                app.UpSlider.Value=app.DownSlider.Value-cycle;
            else
                app.UIAxes.XLim=[value, app.DownSlider.Value];
            end
        end

        % Callback function
        function DownSliderValueChanged(app, event)
         global cycle
             value = app.DownSlider.Value;
             if value<=app.UpSlider.Value
                app.UIAxes.XLim=[app.UpSlider.Value, app.UpSlider.Value+cycle];
             else
                app.UIAxes.XLim=[app.UpSlider.Value, value];
             end
        end

        % Value changed function: FilesListBox
        function FilesListBoxValueChanged(app, event)
         global full_path
         global InputFile
         global time
         global cycle
         global count_cycle
         global start_stim
         global Herz
         global points_mod
         global fname
         global All_group_names
         global num_grps
         global All_dataset_name
         global num_dts
         global fid
    
              
              
              value = app.FilesListBox.Value;
              Items = strtrim(app.FilesListBox.Items);
              im = strtrim(value);
              num_Files = find(strcmp(Items, im))  ;
              fname=app.FilesListBox.Items(num_Files);
              InputFile=full_path(num_Files);
              fid = H5F.open(InputFile);
              time=h5read(InputFile,'/time');
              cycle=h5read(InputFile, '/parameters/stim_period');
              start_stim=h5read(InputFile, '/parameters/stim_start');
              count_cycle=idivide(max(int64(time)),int64(cycle));
              Herz=1000/cycle;
              points_mod=mod(max(int64(time)),int64(cycle));
             
              app.CycleLengthLabel.Text=strcat('Cycle Length: ',num2str(cycle));
              app.FrequencyLabel.Text=strcat('Frequency (Hz): ',num2str(Herz,2));
              app.NumbersofCyclesLabel.Text=strcat('Numbers of Cycles: ',num2str(count_cycle)); 
                      
              a=app.Struct_treedd.Children; 
              a.delete;
          
              idx = 0;
              while true
                   Group_name = H5G.get_objname_by_idx(fid,idx);
                   if isempty(Group_name)
                    break
                   end
                   All_group_names{idx+1,1} = Group_name;  
                   category_group(idx+1) = uitreenode(app.Struct_treedd,'Text',Group_name,'NodeData',[]);
                   idx = idx+1;
              end
                        
              num_grps = length(All_group_names);

              All_dataset_name=strings;
              for i=1:num_grps
                   path_dataset(i)='/'+string(All_group_names(i));%+'/';
                   idx=0;
                   add_new=true;
                   while add_new
                     try
                       Dataset_name=H5L.get_name_by_idx(fid,path_dataset(i),'H5_INDEX_NAME','H5_ITER_INC',idx,'H5P_DEFAULT');
                     catch
                       add_new=false;   
                     end
                     if add_new 
                       All_dataset_name(i,idx+1)=Dataset_name;
                       if All_group_names(i)=="parameters"
                           Dataset_value=h5read(InputFile,strcat('/',string(All_group_names(i)),'/',Dataset_name));
                           Dataset_name=strcat(Dataset_name, ':'," ",num2str(Dataset_value));
                           parameters_index=i;
                       end
                       category_dataset(idx+1)=uitreenode(category_group(i),'Text',Dataset_name);
                       idx=idx+1;
                     end    
                   end
                   num_dts(i)=idx; %number datasets in each group
              end 
              app.CycleLengthLabel.Text=strcat('Cycle Length: ',num2str(cycle));
              app.FrequencyLabel.Text=strcat('Frequency (Hz): ',num2str(Herz,2));
              app.NumbersofCyclesLabel.Text=strcat('Numbers of Cycles: ',num2str(count_cycle));
              
             
              
        end

        % Button pushed function: NewIntDataButton
        function NewIntDataButtonPushed(app, event)
         global InputFile
         global time
         global cycle
         global Herz
         global points_mod
         global Dep_Names
         global fname
         global DataContainer
         global sort_flag_dep
            
             function Integral = Int (Current, coefficient)
                     Integral=0;
                     for h=1:cycle+1
                       Integral=Integral+Current(h)*ht*coefficient;
                     end   
             end
            fname=app.FilesListBox.Value;
            Int_Values=matlab.lang.makeValidName(fname);
            IntVal=char(Int_Values); % variable name of new column
            sort_flag_dep=0;          
            Tb=app.IntTable;
           
            if isempty(Tb.Data)
              Dep_Names={'Frequency (Hz)'};
              Cycle_name={'Cycle length (ms)'};
              CaInt_names={'Int{i_relSR} (mM)';'Int{i_relcyt} (mM)';'Int{i_leak} (mM)'; 'Int{i_up} (mM)';'Int{i_xfercyt} (mM)'; ...
                  'Int{i_bCa} (mM)';'Int{i_pCa} (mM)'; ...
                  'Int{i_NaCain} (mM)'; 'Int{i_NaCaout} (mM)'; 'Int{i_NaCatotal} (mM)'; 'Int{i_CaLcyt} (mM)';...
                  'Int{CaTnC} (mM)'; 'Int{Buffc} (mM)'; 'Int{Cain_total} (mM)'; 'Int{Caout_total} (mM)'; ...
                  'Int{i_CaLSS} (mM)'; 'Int{i_relSS} (mM)'; 'Int{i_xferSS} (mM)';'Int{CainSS_total} (mM)'; 'Int{CaoutSS_total} (mM)' };

              app.SaveInttabletoXLSButton.Enable=1;

              app.XAxisIntDropDown.Items=Dep_Names(1:2);
              app.DrawIntchartButton.Enable=1;
              app.DelColIntDropDown.Enable=1;
              app.DelColIntDropDown.Items={};
              app.DelColIntButton.Enable=1;
            end

            app.NewIntDataButton.Text='Add data' ;
            exColumnNames=transpose(Tb.ColumnName);
             
            newColumnName=char(fname);
            %if ismember (fname{1}, exColumnNames)
            if ismember (fname, exColumnNames)
               uialert(app.MainUIFigure,'Data is already in the table','Existing Data');
            else            
               app.DelColIntDropDown.Items=[app.DelColIntDropDown.Items, {newColumnName}];    
               Tb.ColumnName= [exColumnNames, {newColumnName}];
               
               exColumnWidth=Tb.ColumnWidth;
               newColumnWidth={80};
               Tb.ColumnWidth=[exColumnWidth,  newColumnWidth];
               valuesToAdd=([]);
               valuesToAdd=[double(Herz)];
                
               t=time((max(time)+1-cycle-points_mod):(max(time)+1-points_mod));
               t_beg=time(1:cycle+1);
               ht=t_beg(2)-t_beg(1);
             
               valuesToAdd=[valuesToAdd;double(cycle)];
                
               VjSR=h5read(InputFile,'/parameters/V_jSR');
               Vc=h5read(InputFile,'/parameters/V_c');
               VnSR=h5read(InputFile,'/parameters/V_nSR');
               Vss=h5read(InputFile,'/parameters/V_ss');
               FF=h5read(InputFile,'/parameters/F');
               CC=h5read(InputFile,'/parameters/Cm');
               Kbuffc=h5read(InputFile,'/parameters/K_buf_c');
               BuffC=h5read(InputFile,'/parameters/Buf_c');
            
               irel=h5read(InputFile,'/currents/i_rel',t(1),double(cycle+1));
               % 'Int{i_relSR} (mM)','Int{i_relcyt} (mM)' 'Int{i_relSS} (mM)',
               IntIrelSR=Int(irel,1);
               IntIrelcyt=Int(irel,VjSR/Vc);
               IntIrelSS=Int(irel,VjSR/Vss);
               
               ileak=h5read(InputFile,'/currents/i_leak',t(1),double(cycle+1));
               %'Int{i_leak} (mM)';
               IntIleak=Int(ileak,1);
                            
               iup=h5read(InputFile,'/currents/i_up',t(1),double(cycle+1));
               %'Int{i_up} (mM)'
               IntIup=Int(iup,1);
                
               ixfer=h5read(InputFile,'/currents/i_xfer',t(1),double(cycle+1));
               %ixfer=0;
               %'Int{i_xfercyt} (mM)''Int{i_xferSS} (mM)',
               %IntIxfercyt=0;
               IntIxfercyt=Int(ixfer,1);
               %IntIxferSS=0;
               IntIxferSS=Int(ixfer,Vc/Vss);
                
               ibCa=h5read(InputFile,'/currents/i_b_Ca',t(1),double(cycle+1));
               %ibCa=0;
               %'Int{i_bCa} (mM)',
               %IntIbCa=0;
               IntIbCa=Int(ibCa,CC/(2*Vc*FF));
                
               ipCa=h5read(InputFile,'/currents/i_p_Ca',t(1),double(cycle+1));
               %ipCa=0;
               %'Int{i_pCa} (mM)',
               %IntIpCa=0; 
               IntIpCa=Int(ipCa,CC/(2*Vc*FF));
            
                
               iNaCa=h5read(InputFile,'/currents/i_NaCa',t(1),double(cycle+1));
               %'Int{i_NaCain} (mM)', 'Int{i_NaCaout} (mM)', 'Int{i_NaCatotal} (mM)',
               IntINaCain=0;
               IntINaCaout=0;
               IntINaCatotal=0;
               for k=1:cycle+1
                if iNaCa(k)<0
                  IntINaCaout=IntINaCaout+iNaCa(k)*ht*CC/(Vc*FF)*ht;
                end
                if iNaCa(k)>0
                  IntINaCain=IntINaCain+iNaCa(k)*ht*CC/(Vc*FF)*ht;
                end
                IntINaCatotal=IntINaCatotal+iNaCa(k)*ht*CC/(Vc*FF)*ht;
               end  
            
               iCaL=h5read(InputFile,'/currents/i_CaL',t(1),double(cycle+1));
               %'Int{i_CaLcyt} (mM)''Int{i_CaLSS} (mM)',
               IntICaLcyt=Int(iCaL,CC/(2*Vc*FF));
               IntICaLSS=Int(iCaL,CC/(2*Vss*FF));
               
               CaTnC=h5read(InputFile,'/variables/A',t(1),double(cycle+1));
                %dA/dt
               for j=2:cycle+1
                   dCaTnC(j)=(CaTnC(j)-CaTnC(j-1))/ht;
               end
               dCaTnC(1)=0;
               %'Int{CaTnC} (mM)', 
               IntICaTnc=Int(dCaTnC,1);
                
               Cai=h5read(InputFile,'/variables/Ca_i',t(1),double(cycle+1));
               %dCai/dt
               for j=2:cycle+1
                     dCai(j)=(Cai(j)-Cai(j-1))/ht;
               end
               dCai(1)=0;
               %'Int{Buffc} (mM)',
               IntBuffc=0;
               for i=1:cycle+1
                     IntBuffc=IntBuffc+Kbuffc*BuffC/(Cai(i)+Kbuffc)*dCai(i)*ht;
               end
                
                %'Int{Cain_total} (mM)', 'Int{Caout_total} (mM)', 
                IntCain=-IntICaLcyt+IntINaCain-IntIbCa;
                IntCaout=-IntIpCa+IntINaCaout;
                 
                %Int{CainSS_total} (mM)', 'Int{CaoutSS_total} (mM)'
                IntCainSS=-IntICaLSS+IntIrelSS;
                IntCaoutSS=-IntIxferSS;

                valuesToAdd=[valuesToAdd; IntIrelSR; IntIrelcyt; IntIleak; IntIup; IntIxfercyt;  IntIbCa; IntIpCa; ... 
                    IntINaCain;  IntINaCaout;  IntINaCatotal; IntICaLcyt; ...
                    IntICaTnc; IntBuffc;  IntCain; IntCaout; ...
                    IntICaLSS; IntIrelSS; IntIxferSS; IntCainSS; IntCaoutSS]; 
                
                DataContainer=struct(IntVal,valuesToAdd);
               
                Tb.Data=[Tb.Data array2table(DataContainer.(IntVal), 'VariableNames', {IntVal})];     
            end    
        end

        % Button pushed function: DelListButton
        function DelListButtonPushed(app, event)
            global n_files
            global full_path
            
              % Проверяем, есть ли выделенный элемент
                if ~isempty(app.FilesListBox.Value)
                    % Запоминаем текущий индекс выделенного элемента
                    currentIdx = find(strcmp(app.FilesListBox.Items, app.FilesListBox.Value));
                    
                    % Удаляем выбранный элемент из списка
                    app.FilesListBox.Items(currentIdx) = [];
                    
                    % Если используем ItemsData
                    if ~isempty(app.FilesListBox.ItemsData)
                        app.FilesListBox.ItemsData(currentIdx) = [];
                    end
                    
                    full_path(currentIdx)=[];
                    n_files=n_files-1;
                    
                    
                    % Определяем новый элемент для выбора (логика выбора следующего/предыдущего)
                    if ~isempty(app.FilesListBox.Items)
                        
                        % Если удалили не последний элемент - выбираем следующий
                        if currentIdx <= numel(app.FilesListBox.Items)
                            newSelection = app.FilesListBox.Items{currentIdx};
                        else
                            % Если удалили последний - выбираем предыдущий
                            newSelection = app.FilesListBox.Items{end};
                        end
                        
                        app.FilesListBox.Value = newSelection;
                         % Вызываем обработчик изменения выбора (он сам обновит дерево)
                        FilesListBoxValueChanged(app, event);
                    else
                        % Если список стал пустым - сбрасываем выделение
                        app.FilesListBox.Value = {};
                        % Проверяем, есть ли узлы
                        if ~isempty(app.Struct_treedd.Children)
                            % Удаляем все корневые узлы (и их потомки автоматически)
                            delete(app.Struct_treedd.Children);
                            
                            % Альтернатива: app.Struct_treedd.Children = [];
                        end
                        
                        % Обновляем состояние дерева
                        drawnow;  % Для немедленного отображения изменений
                       
                        app.CycleLengthLabel.Text=strcat('Cycle Length: ');
                        app.FrequencyLabel.Text=strcat('Frequency (Hz): ');
                        app.NumbersofCyclesLabel.Text=strcat('Numbers of Cycles: ');
                    end
                    
                   
                    
                    % Обновляем состояние кнопки Delete
                     app.DelListButton.Enable = ~isempty(app.FilesListBox.Items);
                end
        end

               legendTable.LineWidth(i) = line.LineWidth;

        % Button pushed function: AddlinesButton
        function AddlinesButtonPushed(app, event)
            global size_choice
            global text_selected
            global InputFile
            global cycle
            global points_mod
            global Group_selected
            
             title_selected_file=strrep(app.FilesListBox.Value,'_', ' ');
             legend_name=strcat(title_selected_file);
             
             
              % 1. Явно создаём легенду (если ещё не создана)
            if isempty(app.ax(8).Legend)
                legend(app.ax(8), 'show');
            end
            
            % 2. Получаем все объекты с DisplayName
            lines = findobj(app.ax(8), 'Type', 'line', '-property', 'DisplayName');
            
            % 3. Собираем все имена
            legend_names = {};
            for i = 1:length(lines)
                if ~isempty(lines(i).DisplayName)
                    legend_names{end+1} = lines(i).DisplayName;
                end
            end
             
            if ismember(legend_name, legend_names)
                disp(['Ряд "' legend_name '" уже существует']);
                uialert(app.MainUIFigure, 'Ряд уже существует!', 'Ошибка');
                % Действия если имя уже существует
            else
             for idx=1:size_choice 
                time=h5read(InputFile,'/time');
                t=time((max(time)+1-cycle-points_mod):(max(time)+1-points_mod));
                t_sdvig=t-t(1);
                path_dataset=strcat('/',Group_selected(idx),'/',text_selected(idx));
                vars=h5read(InputFile,path_dataset,t(1),double(cycle+1));
                
                if contains(text_selected(idx),"_")
                 title_selected_dataset=strcat(extractBefore(text_selected(idx),"_"),"_{",extractAfter(text_selected(idx),"_"),"}");
                else
                 title_selected_dataset=text_selected(idx);   
                end  
            
      
               if  app.ax(idx).YLim(1)>min(vars)
                   app.ax(idx).YLim(1)=min(vars);
               end
               if app.ax(idx).YLim(2)<max(vars)
                  app.ax(idx).YLim(2)=max(vars); 
               end
                if app.ax(idx).XLim(2)<max(time)-t(1)+1-points_mod
                   app.ax(idx).XLim(2)=max(time)-t(1)+1-points_mod; 
                end  
                
              hold(app.ax(idx), 'on');
              hold(app.ax(8),'on');
             
              
              plot(app.ax(idx),t_sdvig,vars);%, "DisplayName",legend_name);
           
              %legend(app.ax(idx), 'show');
              end
              plot(app.ax(8),NaN,NaN, "DisplayName",legend_name);
            end   
        end

        % Value changed function: CheckStimulBox
        function CheckStimulBoxValueChanged(app, event)
        global cycle
        global count_cycle         
        global vars
        global start_stim
        
            value = app.CheckStimulBox.Value;
            if value==1
              hold(app.UIAxes, 'on');
              highlight_indices = [];
              
              for cycle_num= 1:count_cycle
                  start_idx = (cycle_num - 1) * cycle + 1;
                  target_idx = start_idx + start_stim-1;
                  if target_idx <= length(vars)
                    highlight_indices = [highlight_indices, target_idx];
                  end
              end
              plot(app.UIAxes, highlight_indices, vars(highlight_indices), '^r', 'MarkerSize', 8, 'MarkerFaceColor', 'r', "DisplayName",'Stimuli');
              app.DropDown.Items=[app.DropDown.Items, 'Stimuli'];
              if app.modeSwitch.Value==0
                  hold(app.UIAxes, 'off');
              end
            end
            if value==0
             h_stimuli = findobj(app.UIAxes, 'DisplayName', 'Stimuli');
             if ~isempty(h_stimuli)
                delete(h_stimuli);
             end
             
            current_items = app.DropDown.Items;
            
            
            % Находим индекс элемента 'Stimuli'
            idx_to_remove = find(strcmp(current_items, 'Stimuli'));
            
            if ~isempty(idx_to_remove)
                % Удаляем элемент из списка
                app.DropDown.Items(idx_to_remove) = [];
                

                % 3. Проверяем, сколько элементов осталось
                if isempty(app.DropDown.Items)
                    % Случай 1: НЕ ОСТАЛОСЬ НИ ОДНОГО ЭЛЕМЕНТА
                    app.DropDown.Enable = 'off';
                    app.DropDown.Value = ''; % очищаем значение
                    app.SeriesclearButton.Enable = 'off';
                    app.ClearChartButton.Enable = 'off';
                else
                    % Случай 2: ОСТАЛОСЬ НЕСКОЛЬКО ЭЛЕМЕНТОВ (включая возможность одного)
                    app.DropDown.Enable = 'on';
                    app.DropDown.Value = app.DropDown.Items{1}; % выбираем первый элемент
                    app.SeriesclearButton.Enable = 'on';
                    app.ClearChartButton.Enable = 'on';
                end
            end
            end
        end
    end

    % App creation and deletion
    methods (Access = public)

        % Construct app
        function app = My_H5_reader

            % Create UIFigure and components
            createComponents(app)

            % Register the app with App Designer
            registerApp(app, app.MainUIFigure)

            % Execute the startup function
            runStartupFcn(app, @startupFcn)

            if nargout == 0
                clear app
            end
        end

        % Code that executes before app deletion
        function delete(app)

            % Delete UIFigure when app is deleted
            delete(app.MainUIFigure)
        end
    end
